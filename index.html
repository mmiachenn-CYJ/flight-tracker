// ════════════════════════════════════════════════════ // Flight Tracker — Cloudflare Worker // 金鑰全部存在環境變數，前端完全看不到 // 需要的環境變數（類型都選 Text）： // SERPAPI_KEY = 你的 SerpApi Key // ANTHROPIC_KEY = 你的 Anthropic Key // ════════════════════════════════════════════════════ const CORS = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Access-Control-Max-Age': '86400', }; function json(obj, status = 200) { return new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json', ...CORS }, }); } export default { async fetch(request, env) { // 最外層 try/catch：確保任何錯誤都帶 CORS 回傳， // 避免瀏覽器看到沒有 CORS 的錯誤而報「Failed to fetch」 try { return await handle(request, env); } catch (e) { return json({ error: 'Worker 例外：' + (e && e.message ? e.message : String(e)) }, 500); } }, }; async function handle(request, env) { // CORS 預檢 if (request.method === 'OPTIONS') { return new Response(null, { headers: CORS }); } if (request.method !== 'POST') { return json({ error: 'Only POST allowed' }, 405); } let body; try { body = await request.json(); } catch { return json({ error: 'Invalid JSON' }, 400); } const action = body && body.action; // ── 動作 0：連線/環境變數自我檢測（不洩漏 Key 內容）── if (action === 'ping') { return json({ ok: true, pong: true, has_serpapi: !!env.SERPAPI_KEY, has_anthropic: !!env.ANTHROPIC_KEY, time: new Date().toISOString(), }); } // ── 動作 1：查機票票價 ── if (action === 'search_flights') { return await searchFlights(body, env); } // ── 動作 2：AI 分析趨勢 ── if (action === 'ai_analysis') { return await aiAnalysis(body, env); } return json({ error: 'Unknown action: ' + action }, 400); } // ──────────────────────────────────────────────────── // 查機票票價（透過 SerpApi → Google Flights） // ──────────────────────────────────────────────────── async function searchFlights(body, env) { const { origin, dest, outbound_date, return_date, trip_type } = body; if (!env.SERPAPI_KEY) { return json({ error: '環境變數 SERPAPI_KEY 未設定' }, 500); } if (!origin || !dest || !outbound_date) { return json({ error: '缺少必要參數 origin / dest / outbound_date' }, 400); } const type = trip_type === 'round_trip' ? '1' : '2'; const params = new URLSearchParams({ engine: 'google_flights', departure_id: origin, arrival_id: dest, outbound_date, currency: 'TWD', hl: 'zh-tw', gl: 'tw', type, api_key: env.SERPAPI_KEY, }); if (type === '1' && return_date) params.set('return_date', return_date); const url = 'https://serpapi.com/search.json?' + params.toString(); let data; try { const resp = await fetch(url); data = await resp.json(); } catch (e) { return json({ error: 'SerpApi 連線失敗：' + e.message }, 502); } if (data.error) { return json({ error: 'SerpApi: ' + data.error }, 502); } const flights = [...(data.best_flights || []), ...(data.other_flights || [])]; const prices = flights.map(f => f.price).filter(p => typeof p === 'number'); const lowest = prices.length ? Math.min(...prices) : null; const insights = data.price_insights || {}; return json({ ok: true, lowest_price: lowest, price_level: insights.price_level || null, lowest_seen: insights.lowest_price || null, typical_range: insights.typical_price_range || null, price_history: insights.price_history || null, currency: 'TWD', fetched_at: new Date().toISOString(), }); } // ──────────────────────────────────────────────────── // AI 分析（透過 Claude API） // ──────────────────────────────────────────────────── async function aiAnalysis(body, env) { const { route_label, history, typical_range, days_until_departure } = body; if (!env.ANTHROPIC_KEY) { return json({ error: '環境變數 ANTHROPIC_KEY 未設定' }, 500); } const prompt = `你是一位幫旅客判斷機票購買時機的分析師，說話像朋友、白話、不用術語。請用繁體中文。 航線：${route_label} 最近票價走勢（新到舊，單位 TWD）：${JSON.stringify(history)} 去年同期常態價格區間：${typical_range ? `${typical_range[0]} ~ ${typical_range[1]}` : '無資料'} 距離出發還有：${days_until_departure != null ? days_until_departure + ' 天' : '未知'} 請只輸出以下 JSON 格式，不要任何其他文字： { "summary": "用 2~3 句話說明目前票價走勢與所處水位", "verdict": "buy_now" 或 "wait" 或 "neutral", "verdict_text": "一句話的具體建議，例如『建議盡快購買』或『可再觀望』" }`; let data; try { const resp = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-api-key': env.ANTHROPIC_KEY, 'anthropic-version': '2023-06-01', }, body: JSON.stringify({ model: 'claude-sonnet-4-6', max_tokens: 600, messages: [{ role: 'user', content: prompt }], }), }); data = await resp.json(); } catch (e) { return json({ error: 'Claude 連線失敗：' + e.message }, 502); } if (data.error) { return json({ error: 'Claude: ' + (data.error.message || 'unknown'), claude_error_type: data.error.type || null, }, 502); } let text = ''; if (Array.isArray(data.content)) { text = data.content.filter(b => b.type === 'text').map(b => b.text).join(''); } if (!text) { return json({ error: 'Claude 沒有回傳文字內容', raw: JSON.stringify(data).slice(0, 500) }, 502); } let parsed; try { parsed = JSON.parse(text.replace(/```json|```/g, '').trim()); } catch { parsed = { summary: text, verdict: 'neutral', verdict_text: '' }; } return json({ ok: true, ...parsed }); }
